Đối soát thanh toán

Kênh nào có chênh lệch thực nhận so với dự kiến?

SELECT o.channel,COUNT(*) AS transactions,
 SUM(CASE WHEN p.actual_payout_vnd<>p.expected_payout_vnd THEN 1 ELSE 0 END) AS discrepant_transactions,
 SUM(p.expected_payout_vnd) AS expected_payout_vnd,SUM(p.actual_payout_vnd) AS actual_payout_vnd,
 SUM(p.actual_payout_vnd-p.expected_payout_vnd) AS net_difference_vnd,
 SUM(ABS(p.actual_payout_vnd-p.expected_payout_vnd)) AS absolute_difference_vnd,
 SUM(CASE WHEN p.actual_payout_vnd<p.expected_payout_vnd THEN p.expected_payout_vnd-p.actual_payout_vnd ELSE 0 END) AS shortfall_vnd
FROM payments p JOIN orders o ON o.order_id=p.order_id
WHERE o.order_date>='2026-01-01' AND o.order_date<'2026-04-01'
GROUP BY o.channel ORDER BY absolute_difference_vnd DESC;

Đối soát tồn kho

Tồn sổ và kiểm kê ngày 31/03 chênh lệch như thế nào?

Tồn sổ đã gồm OPENING và RECEIPT trong movements; không cộng receipts lần nữa. Thiếu kiểm kê giữ NULL, không thay bằng 0.

Xem câu lệnh SQL
WITH stock AS (
 SELECT sku,warehouse_id,SUM(qty_signed) AS book_qty FROM inventory_movements
 WHERE DATE(date)<='2026-03-31' GROUP BY sku,warehouse_id
), counts AS (
 SELECT sku,warehouse_id,counted_qty FROM stock_counts WHERE DATE(count_date)='2026-03-31'
), universe AS (
 SELECT sku,warehouse_id FROM stock UNION SELECT sku,warehouse_id FROM counts
)
SELECT u.sku,u.warehouse_id,p.product_name,COALESCE(s.book_qty,0) AS book_qty,
 c.counted_qty,c.counted_qty-COALESCE(s.book_qty,0) AS variance_qty,
 (c.counted_qty-COALESCE(s.book_qty,0))*p.standard_cost_vnd AS variance_value_vnd,
 CASE WHEN c.counted_qty IS NULL THEN 'NO_COUNT'
 WHEN c.counted_qty<>COALESCE(s.book_qty,0) THEN 'MISMATCH' ELSE 'MATCH' END AS check_status
FROM universe u LEFT JOIN stock s ON s.sku=u.sku AND s.warehouse_id=u.warehouse_id
LEFT JOIN counts c ON c.sku=u.sku AND c.warehouse_id=u.warehouse_id
LEFT JOIN products p ON p.sku=u.sku
ORDER BY ABS(c.counted_qty-COALESCE(s.book_qty,0)) DESC,u.sku,u.warehouse_id;

Tồn cuối ngày liên tục

Làm sao tạo lịch tồn kho kể cả ngày không phát sinh?

Xem câu lệnh SQL
WITH RECURSIVE dates(date) AS (
 SELECT '2026-01-01' UNION ALL SELECT DATE(date,'+1 day') FROM dates WHERE date<'2026-03-31'
), warehouses AS (
 SELECT DISTINCT warehouse_id FROM inventory_movements
), daily AS (
 SELECT DATE(date) AS date,sku,warehouse_id,SUM(qty_signed) AS movement_qty FROM inventory_movements
 WHERE DATE(date)>='2026-01-01' AND DATE(date)<'2026-04-01' GROUP BY DATE(date),sku,warehouse_id
), opening AS (
 SELECT sku,warehouse_id,SUM(qty_signed) AS opening_qty FROM inventory_movements WHERE DATE(date)<'2026-01-01' GROUP BY sku,warehouse_id
), grid AS (
 SELECT d.date,p.sku,w.warehouse_id,COALESCE(m.movement_qty,0) AS movement_qty,COALESCE(o.opening_qty,0) AS opening_qty
 FROM dates d CROSS JOIN products p CROSS JOIN warehouses w

FROM grid ORDER BY sku,warehouse_id,date;

Năng suất kho

Năng suất và tỷ lệ lỗi đóng gói của từng nhân viên như thế nào?

Tổng hợp công và tác vụ riêng trước khi JOIN. Có tính ngày chấm công không có task. Giờ khả dụng khác thời gian xử lý thực tế.

Xem câu lệnh SQL
WITH task_day AS (
 SELECT DATE(date) AS date,employee_id,warehouse_id,COUNT(*) AS pack_tasks,SUM(units) AS units,SUM(error_flag) AS error_tasks
 FROM warehouse_tasks WHERE task_type='PACK' AND date>='2026-01-01' AND DATE(date)<'2026-04-01'
 GROUP BY DATE(date),employee_id,warehouse_id
), attendance_day AS (
 SELECT DATE(date) AS date,employee_id,warehouse_id,MAX(employee_name) AS employee_name,SUM(paid_hours-break_hours) AS available_hours
 FROM attendance WHERE department='Kho' AND date>='2026-01-01' AND DATE(date)<'2026-04-01'
 GROUP BY DATE(date),employee_id,warehouse_id
), combined AS (
 SELECT a.*,COALESCE(t.pack_tasks,0) AS pack_tasks,COALESCE(t.units,0) AS units,COALESCE(t.error_tasks,0) AS error_tasks
 FROM attendance_day a LEFT JOIN task_day t ON t.date=a.date AND t.employee_id=a.employee_id AND t.warehouse_id=a.warehouse_id
)
SELECT employee_id,MAX(employee_name) AS employee_name,SUM(available_hours) AS available_hours,
 SUM(pack_tasks) AS pack_tasks,SUM(units) AS packed_units,
 ROUND(1.0*SUM(units)/NULLIF(SUM(available_hours),0),2) AS units_per_available_hour,
 ROUND(100.0*SUM(error_tasks)/NULLIF(SUM(pack_tasks),0),2) AS error_rate_pct
FROM combined GROUP BY employee_id ORDER BY units_per_available_hour DESC,employee_id;

Hiệu quả quảng cáo video

Video nào tạo ROAS cao và chi phí sản xuất bao nhiêu?

ROAS = tổng doanh thu do nền tảng gán / tổng spend; không AVG ROAS từng ngày. Chi phí video chỉ một lần. ROAS không chứng minh hiệu quả tăng thêm.

Xem câu lệnh SQL
WITH ads AS (
 SELECT video_id,SUM(spend_vnd) AS spend_vnd,SUM(impressions) AS impressions,SUM(clicks) AS clicks,
 SUM(platform_attributed_orders) AS attributed_orders,SUM(platform_attributed_revenue_vnd) AS attributed_revenue_vnd
 FROM ads_daily WHERE DATE(date)>='2026-01-01' AND DATE(date)<'2026-04-01' GROUP BY video_id
)
SELECT a.video_id,c.creator_id,c.topic,a.spend_vnd,a.attributed_revenue_vnd,
 ROUND(1.0*a.attributed_revenue_vnd/NULLIF(a.spend_vnd,0),2) AS roas,
 ROUND(100.0*a.clicks/NULLIF(a.impressions,0),2) AS ctr_pct,
 ROUND(1.0*a.spend_vnd/NULLIF(a.attributed_orders,0),0) AS attributed_cpa_vnd,
 CASE WHEN c.publish_date>='2026-01-01' AND c.publish_date<'2026-04-01' THEN c.production_cost_vnd ELSE 0 END AS q1_production_cost_vnd
FROM ads a LEFT JOIN content c ON c.video_id=a.video_id ORDER BY roas DESC,a.video_id;

Đóng góp sau Ads và nội dung

Sau phí, quảng cáo và sản xuất video còn lại bao nhiêu?

Đây là đóng góp của tháng, không phải lợi nhuận ròng hoặc đo lường tác động nhân quả của Ads. Chưa có lương, thuê kho, thuế và toàn bộ OPEX.

Xem câu lệnh SQL
WITH sales AS (
 SELECT SUBSTR(DATE(order_date),1,7) AS month,SUM(net_sales_vnd) AS net_sales_vnd,SUM(contribution_before_ads_vnd) AS contribution_before_ads_vnd
 FROM vw_orders_kpi WHERE DATE(order_date)>='2026-01-01' AND DATE(order_date)<'2026-04-01' GROUP BY 1
), ads AS (
 SELECT SUBSTR(DATE(date),1,7) AS month,SUM(spend_vnd) AS spend_vnd FROM ads_daily
 WHERE DATE(date)>='2026-01-01' AND DATE(date)<'2026-04-01' GROUP BY 1
), production AS (
 SELECT SUBSTR(DATE(publish_date),1,7) AS month,SUM(production_cost_vnd) AS production_cost_vnd FROM content
 WHERE DATE(publish_date)>='2026-01-01' AND DATE(publish_date)<'2026-04-01' GROUP BY 1
)
SELECT s.*,COALESCE(a.spend_vnd,0) AS spend_vnd,COALESCE(p.production_cost_vnd,0) AS production_cost_vnd,
 s.contribution_before_ads_vnd-COALESCE(a.spend_vnd,0)-COALESCE(p.production_cost_vnd,0) AS contribution_after_ads_content_vnd
FROM sales s LEFT JOIN ads a ON a.month=s.month LEFT JOIN production p ON p.month=s.month ORDER BY s.month;

Quy tắc và giới hạn phân tích

  • Đơn hủy: doanh thu và giá vốn bằng 0. Đơn hoàn: hoàn toàn bộ tiền hàng, nhập lại toàn bộ hàng; chi phí giao hàng vẫn phát sinh.
  • Doanh thu thuần = số lượng × đơn giá − giảm giá dòng − hoàn tiền dòng, áp dụng đơn không hủy.
  • Giá vốn chỉ tính cho đơn completed. Giá vốn của mô hình cố định theo SKU, chưa phải FIFO thực tế.
  • Phí sàn và vận chuyển giữ cấp đơn; doanh thu và giá vốn tổng hợp từ dòng hàng trước khi ghép về đơn.
  • Ads attributed revenue là chỉ số do nền tảng gán theo cửa sổ 7 ngày click và 1 ngày view, không cộng thêm vào doanh thu nội bộ.
  • Chi phí nội dung ghi nhận một lần/video theo publish_date. Nội dung chưa chạy Ads vẫn được tính trong tổng chi phí nội dung.
  • Phân tích thanh toán theo cohort đơn Q1, không giới hạn settlement_date vào Q1. Phân tích theo ngày thanh toán sẽ có phạm vi khác.
  • Chưa có lý do hủy/hoàn, toàn bộ chi phí vận hành, dữ liệu target hay thiết kế thực nghiệm để kết luận nguyên nhân.
